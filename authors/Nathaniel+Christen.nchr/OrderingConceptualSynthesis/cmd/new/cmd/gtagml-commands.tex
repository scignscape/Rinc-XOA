%\newcommand{\@sect}[1]{\section{#1}}

\newcommand{\sL@section}[1]{\vspace{-2pt}\section{#1}\vspace{-2pt}}
\newcommand{\sL@subsection}[1]{\vspace{14pt}\subsection{#1}

\vspace{-18pt}
\textcolor[rgb]{0.6,0.6,0.8}{\rule{\columnwidth}{0.4pt}\vspace{-15pt}}}

\newcommand{\@sectionLevel}[2]{%
\ifcase#1\relax
\or{\sL@section{#2}}
\or{\sL@subsection{#2}}
\else{}
\fi
}


\NewDocumentCommand{\s}{E{.}{}D||{}m}{%
\IfNoValueTF{#1}{
\@sectionLevel{#2}{#3}
}
{%else
\@sectionLevel{#1}{#3}
}
}

\newcommand{\@paLevel}[2]{%

\fi
}

}
